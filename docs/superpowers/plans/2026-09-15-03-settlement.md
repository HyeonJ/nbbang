```ts
import { neon } from '@neondatabase/serverless';
import { readFileSync, readdirSync } from 'node:fs';
import { createHash } from 'node:crypto';

// drizzle-kit migrate는 __drizzle_migrations(hash, created_at)로 적용 이력을 본다.
// 이미 스키마가 있는 DB에 baseline을 "실행 없이 적용됨"으로 표시한다.
const sql = neon(process.env.DATABASE_URL!);
const journal = JSON.parse(readFileSync('drizzle/meta/_journal.json', 'utf8')) as {
  entries: { tag: string; when: number }[];
};
await sql`create schema if not exists drizzle`;
await sql`create table if not exists drizzle.__drizzle_migrations (
  id serial primary key, hash text not null, created_at bigint
)`;
for (const e of journal.entries) {
  const body = readFileSync(`drizzle/${e.tag}.sql`, 'utf8');
  const hash = createHash('sha256').update(body).digest('hex');
  const [{ n }] = await sql`select count(*)::int as n from drizzle.__drizzle_migrations where hash = ${hash}`;
  if (n > 0) { console.log(`skip ${e.tag} (already recorded)`); continue; }
  await sql`insert into drizzle.__drizzle_migrations (hash, created_at) values (${hash}, ${e.when})`;
  console.log(`marked ${e.tag}`);
}
```
세 브랜치에 각각 실행:
```powershell

npx dotenv -e .env.test  -- npx tsx scripts/mark-migration-applied.mts
```

### Task 2: 복합 FK 하드닝 (회고 후속 3 — 공개 장부 전에)

**이유:** 지금 DB는 `dues_payments.membership_id`에 **타 모임 멤버십**을 넣는 것을 막지 못한다(Plan 02 Task 9가 그 방법으로 음수 미납 상태를 재현했다). `ledger_entries.reversal_of`도 존재하지 않는 id나 타 모임 엔트리를 가리킬 수 있다. 액션 레이어가 막고 있지만, 공개 장부가 링크로 데이터를 노출하기 시작하면 DB 레벨 불변식이 필요하다.

**Files:** Modify `lib/db/schema.ts`, Create `drizzle/0001_*.sql`(generate 산출물)

- [ ] **Step 1: 복합 유니크 추가** — 복합 FK가 참조할 대상 키를 먼저 만든다. `lib/db/schema.ts`:
```ts
// 복합 FK의 참조 대상 — (group_id, id) 쌍으로 "같은 모임 안의 그 행"을 가리킬 수 있게 한다.
export const memberships = pgTable('memberships', { /* ...기존... */ }, (t) => [
  uniqueIndex('memberships_user_group').on(t.userId, t.groupId),
  unique('memberships_group_id_key').on(t.groupId, t.id),

export const duesRounds = pgTable('dues_rounds', { /* ...기존... */ }, (t) => [
  uniqueIndex('dues_rounds_group_period').on(t.groupId, t.period),
  unique('dues_rounds_group_id_key').on(t.groupId, t.id),
]);

export const ledgerEntries = pgTable('ledger_entries', { /* ...기존... */ }, (t) => [
  unique('ledger_entries_group_id_key').on(t.groupId, t.id),
```

`unique`를 `drizzle-orm/pg-core` import에 추가한다.
- [ ] **Step 2: `dues_payments`에 `groupId` 추가 + 복합 FK 두 개** — 이제 `groupId`는 "두 번째 진실 원천"이 아니라 **복합 FK의 구성요소**다(Plan 02 Task 8 리뷰의 판단: 단독 컬럼만 추가하면 오히려 해롭고, 복합 FK와 함께여야 의미가 있다):
```ts
export const duesPayments = pgTable('dues_payments', {
  id: text('id').primaryKey(),
  groupId: text('group_id').notNull().references(() => groups.id),
  roundId: text('round_id').notNull(),
  membershipId: text('membership_id').notNull(),
  ledgerEntryId: text('ledger_entry_id').notNull(),
  paidAt: timestamp('paid_at', { withTimezone: true }).notNull().defaultNow(),
}, (t) => [
  uniqueIndex('dues_payments_round_membership').on(t.roundId, t.membershipId),
  // 회차·멤버·원장 엔트리가 모두 이 결제와 같은 모임이어야 한다 — DB가 보장한다.
  foreignKey({ columns: [t.groupId, t.roundId], foreignColumns: [duesRounds.groupId, duesRounds.id], name: 'dues_payments_round_fk' }),
  foreignKey({ columns: [t.groupId, t.membershipId], foreignColumns: [memberships.groupId, memberships.id], name: 'dues_payments_membership_fk' }),
  foreignKey({ columns: [t.groupId, t.ledgerEntryId], foreignColumns: [ledgerEntries.groupId, ledgerEntries.id], name: 'dues_payments_entry_fk' }),
]);
```
`foreignKey`를 import에 추가한다. 기존 단일 FK(`references`)는 복합 FK가 대체하므로 제거한다.
- [ ] **Step 3: `ledger_entries.reversal_of`를 같은 모임으로 제한**:
```ts
  // 역분개 대상은 반드시 같은 모임의 엔트리다 — 액션 레이어에 더해 DB도 막는다.
  foreignKey({ columns: [t.groupId, t.reversalOf], foreignColumns: [t.groupId, t.id], name: 'ledger_entries_reversal_fk' }),
```

```markdown
# ADR-003: 정산(엔빵)은 원장에 기록하지 않는다

2026-09-15 · 상태: 승인

## 결정

이벤트 정산(회식비 엔빵 등)의 계산 결과는 `settlements`/`settlement_shares`에만 저장하고
`ledger_entries`에는 **아무것도 넣지 않는다**. 모임 잔액은 정산에 영향받지 않는다.

## 이유

- 원장은 **모임 돈**의 출입을 기록한다(회비 수입, 모임 지출). 정산은 **멤버 사이의 채무** 정리다 —
  민지가 철수에게 8,000원을 보내는 일은 모임 통장을 지나가지 않는다. 이를 원장에 넣으면
  잔액이 모임 자산을 더 이상 뜻하지 않게 되고, ADR-001이 지키려는 "잔액 = 모임이 가진 돈"이 깨진다.
- 정산을 원장에 넣으면 모든 정산이 합계 0인 엔트리 쌍을 만들어 원장을 소음으로 채운다.
- 모임 통장에서 나간 회식비는 **지출(F3)** 로 따로 기록하면 된다. 그 지출을 참여자끼리 어떻게
  나눴는지가 정산이다 — 두 기록의 관심사가 다르다.

## 결과

- 정산 결과는 **스냅샷**이다. 나중에 명단이나 금액이 바뀌어도 과거 정산은 재계산하지 않는다.
- 정산에는 역분개가 없다. 잘못 만든 정산은 삭제하거나 새로 만든다(원장 불변식과 무관하므로).
- 공개 장부(F5)는 원장을 보여주므로 정산은 기본적으로 포함하지 않는다. 필요하면 별도 섹션으로 둔다.

## 기각한 대안

- **정산을 원장 엔트리로 기록**: 잔액의 의미가 오염된다(위 참조).
- **정산 결과를 저장하지 않고 매번 재계산**: 명단이 바뀌면 과거 정산 결과가 소급 변경돼
  "그때 누가 누구에게 얼마 보냈는지"의 기록이 사라진다.
```

```ts
export class SettlementRuleError extends Error {
  constructor(public code: 'INVALID_AMOUNT' | 'NO_PARTICIPANTS' | 'PAYER_NOT_PARTICIPANT') {
    super(code);
  }
}

export type Share = { membershipId: string; amount: number };
export type Transfer = { from: string; to: string; amount: number };

/**
 * 총액을 참여자 수로 나눈다. 나누어떨어지지 않는 나머지는 **앞에서부터 1원씩** 더 낸다 —
 * 무작위로 흘리면 같은 입력에 다른 결과가 나오고, 버리면 합계가 총액과 어긋난다.
 * 분배 합계 = 총액은 이 함수의 불변식이다.
 */
export function splitEvenly(total: number, participantIds: readonly string[]): Share[] {
  if (participantIds.length === 0) throw new SettlementRuleError('NO_PARTICIPANTS');
  if (!Number.isInteger(total) || total <= 0) throw new SettlementRuleError('INVALID_AMOUNT');
  const base = Math.floor(total / participantIds.length);
  let remainder = total - base * participantIds.length;
  return participantIds.map((membershipId) => {
    const extra = remainder > 0 ? 1 : 0;
    remainder -= extra;
    return { membershipId, amount: base + extra };
  });
}

/**
 * 선결제자(payer)가 총액을 이미 냈다는 전제로, 나머지가 payer에게 보낼 이체 목록을 만든다.
 * 한 사람이 냈으므로 채권자는 payer 한 명뿐이라 이체는 최대 n−1건이고, 상계할 것이 없다.
 * (여러 사람이 나눠 낸 정산은 v1 범위 밖 — 그 경우에만 그리디 상계가 필요해진다.)
 */
export function minimalTransfers(
  shares: readonly Share[],
  payerId: string,
  total: number,
): Transfer[] {
  if (shares.length === 0) throw new SettlementRuleError('NO_PARTICIPANTS');
  if (!shares.some((s) => s.membershipId === payerId)) throw new SettlementRuleError('PAYER_NOT_PARTICIPANT');
  if (shares.reduce((n, s) => n + s.amount, 0) !== total) throw new SettlementRuleError('INVALID_AMOUNT');
  return shares
    .filter((s) => s.membershipId !== payerId && s.amount > 0)
    .map((s) => ({ from: s.membershipId, to: payerId, amount: s.amount }));
}
```

```ts
'use server';
import { z } from 'zod';
import { and, eq, inArray } from 'drizzle-orm';
import { groupActionClient, assertOwner, ActionError } from './clients';
import { revalidatePath } from 'next/cache';
import { db } from '@/lib/db';
import { memberships, settlements, settlementShares } from '@/lib/db/schema';
import { splitEvenly, minimalTransfers, SettlementRuleError } from '@/lib/domain/settlement';

export const createSettlement = groupActionClient
  .inputSchema(
    z.object({
      groupId: z.string(),
      title: z.string().min(1).max(50),
      total: z.number().int().positive().max(100_000_000),
      occurredOn: z.string().regex(/^\d{4}-\d{2}-\d{2}$/),
      payerMembershipId: z.string(),
      participantMembershipIds: z.array(z.string()).min(1).max(100),
    }),
  )
  .action(async ({ parsedInput, ctx }) => {
    assertOwner(ctx.role);

    // 참여자·선결제자가 모두 이 모임의 멤버인지 — 모임 경계는 여기서 그린다(ADR-002).
    const ids = Array.from(new Set([...parsedInput.participantMembershipIds, parsedInput.payerMembershipId]));
    const rows = await db
      .select({ id: memberships.id })
      .from(memberships)
      .where(and(eq(memberships.groupId, ctx.groupId), inArray(memberships.id, ids)));
    if (rows.length !== ids.length) throw new ActionError('NOT_MEMBER');

    let shares, transfers;
    try {
      shares = splitEvenly(parsedInput.total, parsedInput.participantMembershipIds);
      transfers = minimalTransfers(shares, parsedInput.payerMembershipId, parsedInput.total);
    } catch (e) {
      throw e instanceof SettlementRuleError ? new ActionError(e.code) : e;
    }

    const settlementId = crypto.randomUUID();
    const shareByMember = new Map(shares.map((s) => [s.membershipId, s.amount]));
    await db.transaction(async (tx) => {
      await tx.insert(settlements).values({
        id: settlementId,
        groupId: ctx.groupId,
        title: parsedInput.title,
        total: parsedInput.total,
        payerMembershipId: parsedInput.payerMembershipId,
        occurredAt: new Date(`${parsedInput.occurredOn}T03:00:00.000Z`), // KST 정오
        createdBy: ctx.userId,
      });
      if (transfers.length > 0) {
        await tx.insert(settlementShares).values(
          transfers.map((t) => ({
            id: crypto.randomUUID(),
            groupId: ctx.groupId,
            settlementId,
            fromMembershipId: t.from,
            toMembershipId: t.to,
            amount: t.amount,
            shareAmount: shareByMember.get(t.from) ?? 0,
          })),
        );
      }
    });
    revalidatePath(`/groups/${ctx.groupId}/settle`);
    return { settlementId };
  });
```

```ts
export type PublicLedger = {
  groupName: string;
  balance: number;
  entries: { id: string; type: string; amount: number; occurredAt: Date; category: string | null; memo: string | null; reversalOf: string | null }[];
  members: { displayName: string; role: string }[];
};

export async function getPublicLedger(token: string): Promise<PublicLedger | null> {
  // 토큰으로 모임을 찾되, 이름만 가져온다 — inviteToken/publicToken은 select에 넣지 않는다.
  const [group] = await db
    .select({ id: groups.id, name: groups.name })
    .from(groups)
    .where(eq(groups.publicToken, token));
  if (!group) return null;

  const [entries, members] = await Promise.all([
    db.select({
      id: ledgerEntries.id, type: ledgerEntries.type, amount: ledgerEntries.amount,
      occurredAt: ledgerEntries.occurredAt, category: ledgerEntries.category,
      memo: ledgerEntries.memo, reversalOf: ledgerEntries.reversalOf,
    }).from(ledgerEntries).where(eq(ledgerEntries.groupId, group.id))
      .orderBy(desc(ledgerEntries.occurredAt), desc(ledgerEntries.createdAt)),
    // 표시 이름과 역할만 — userId·이메일은 select에 없다.
    db.select({ displayName: memberships.displayName, role: memberships.role })
      .from(memberships).where(eq(memberships.groupId, group.id)).orderBy(asc(memberships.joinedAt)),
  ]);

  return {
    groupName: group.name,
    balance: entries.reduce((n, e) => n + e.amount, 0),
    entries,
    members,
  };
}
```

```ts
      formatDateKst(e.occurredAt), e.type, String(e.amount),
      e.category ?? '', e.memo ?? '', e.reversalOf ?? '',
    ]),
  );
  // BOM — Excel이 UTF-8 한글을 깨뜨리지 않게 한다.
  return new NextResponse(`\uFEFF${csv}`, {
    headers: {
      'Content-Type': 'text/csv; charset=utf-8',
      'Content-Disposition': `attachment; filename="nbbang-${groupId.slice(0, 8)}-${formatDateKst(new Date())}.csv"`,
    },
  });
}
```
`getAllEntries(groupId)`를 `queries.ts`에 추가(limit 없이 전체, 날짜 오름차순 — 장부는 시간순으로 읽는다).
